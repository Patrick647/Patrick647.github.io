{{/* Display author list. */}} {{- $authors := .Params.authors -}} {{ if $authors }} {{- if not (reflect.IsSlice $authors) -}} {{- $authors = slice $authors -}} {{- end -}} {{ range $index, $raw := $authors }} {{- $slug := $raw | urlize -}} {{- $profile := partial "functions/get_author_profile" (dict "slug" $slug "title" $raw) -}} {{- $termPage := site.GetPage (printf "/authors/%s" $slug) -}} {{- $termURL := "" -}} {{- with $termPage -}} {{- with .RelPermalink -}}{{- $termURL = . -}}{{- end -}} {{- end -}} {{- $display := $profile.title | default $raw -}} {{- $highlight_name := false -}} {{- range $key, $author := site.Data.authors -}} {{- if $author.is_owner -}} {{- $ownerName := $author.name.display | default ($key | humanize | title) -}} {{- if eq $raw $ownerName -}} {{- $highlight_name = true -}} {{- end -}} {{- end -}} {{- end -}} {{- if gt $index 0 }}, {{ end -}} {{- if $termURL -}} {{ $display }} {{- else -}} {{ $display }} {{- end -}} {{- if isset $.Params "author_notes" -}} {{- with (index $.Params.author_notes $index) -}} {{- partial "author_note_marker" . -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}}
