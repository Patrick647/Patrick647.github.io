{{- /* Renders a multi-line CV summary: one paragraph per line, a leading "Label:" is emphasised. */ -}}
{{- range split (strings.TrimSpace (printf "%v" .)) "\n" -}} {{- $line := strings.TrimSpace . -}} {{- if $line -}} {{- $m := findRESubmatch `^([A-Z][A-Za-z ]{1,24}):\s+` $line 1 -}} {{- if $m -}} {{- $match := index $m 0 -}}

{{ index $match 1 }}: {{ strings.TrimPrefix (index $match 0) $line | markdownify }}

{{- else -}}

{{ $line | markdownify }}

{{- end -}} {{- end -}} {{- end -}}
