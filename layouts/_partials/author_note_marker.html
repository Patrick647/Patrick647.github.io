{{- /* Input: an author_notes entry. Output: a superscript marker, or nothing. */ -}} {{- $note := . -}} {{- $lc := lower $note -}} {{- $marker := "‡" -}} {{- if in $lc "corresponding" -}} {{- $marker = "*" -}} {{- else if or (in $lc "equal") (in $lc "co-first") -}} {{- $marker = "†" -}} {{- else if eq $lc "first author" -}} {{- /* Byline position already shows first authorship. */ -}} {{- $marker = "" -}} {{- end -}} {{- with $marker -}} {{ . }} ({{ $note }}) {{- end -}}
